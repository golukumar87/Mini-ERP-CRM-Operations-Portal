# Run commands for Mini ERP + CRM Portal

# 1. Go to project folder
cd /d "c:\Users\Ravish Kumar\Desktop\React Full Coures\Placement Inforcment"

npm run install:all

# 3. Start backend server
npm run dev --workspace server

# 4. Start frontend (open another terminal)
npm run dev --workspace client

# 5. Open in browser
http://localhost:3000

# 6. Backend API
http://localhost:5000

# 7. Run both together from root (optional)
npm run dev

# 8. Build for production
npm run build --workspace client
npm run build --workspace server
